cmake_minimum_required(VERSION 3.16)
project(OpenMagFDM3D VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(yaml-cpp REQUIRED)
find_package(OpenMP)
add_library(omfdm3d core.cpp current.cpp io.cpp ../tinyexpr/tinyexpr.cpp)
target_include_directories(omfdm3d PUBLIC . .. ../amgcl)
target_compile_definitions(omfdm3d PRIVATE AMGCL_NO_BOOST)
target_link_libraries(omfdm3d PUBLIC Eigen3::Eigen yaml-cpp::yaml-cpp)
if(OpenMP_CXX_FOUND)
  target_link_libraries(omfdm3d PUBLIC OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(omfdm3d PUBLIC /utf-8 /fp:precise)
else()
  target_compile_options(omfdm3d PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(MagFDM3D main.cpp)
target_link_libraries(MagFDM3D PRIVATE omfdm3d)
if(BUILD_TESTING)
  add_executable(solver3d_tests tests.cpp)
  target_link_libraries(solver3d_tests PRIVATE omfdm3d)
  add_test(NAME solver3d_foundation COMMAND solver3d_tests ${CMAKE_CURRENT_BINARY_DIR}/test-artifacts)
endif()
